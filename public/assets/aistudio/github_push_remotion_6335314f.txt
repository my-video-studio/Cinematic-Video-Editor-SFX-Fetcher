this .yml file worked without any error for other project that how you set concurrency at 2x:
name: Render Remotion Video

on:
  workflow_dispatch:
    inputs:
      composition_id:
        description: 'Remotion Composition ID to render (HistoricalDocumentary, HistoricalDocumentaryPart1, or HistoricalDocumentaryPart2)'
        required: true
        default: 'HistoricalDocumentary'
        type: string
      output_filename:
        description: 'Output MP4 filename inside out/'
        required: true
        default: 'video.mp4'
        type: string

jobs:
  render:
    name: Render Video in Cloud
    runs-on: ubuntu-latest
    timeout-minutes: 120

    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Setup Node.js 22
        uses: actions/setup-node@v4
        with:
          node-version: '22'

      - name: Setup Bun
        uses: oven-sh/setup-bun@v2
        with:
          bun-version: latest

      - name: Cache Bun dependencies
        uses: actions/cache@v4
        with:
          path: ~/.bun/install/cache
          key: ${{ runner.os }}-bun-${{ hashFiles('bun.lock') }}
          restore-keys: |
            ${{ runner.os }}-bun-

      - name: Install dependencies (frozen lockfile)
        run: bun install --frozen-lockfile

      - name: Install Linux Chromium shared libraries
        run: |
          sudo apt-get update
          sudo apt-get install -y \
            libnss3 \
            libatk1.0-0 \
            libatk-bridge2.0-0 \
            libcups2 \
            libdrm2 \
            libxkbcommon0 \
            libxcomposite1 \
            libxdamage1 \
            libxfixes3 \
            libxrandr2 \
            libgbm1 \
            libpango-1.0-0 \
            libcairo2 \
            libasound2t64 \
            ffmpeg

      - name: Ensure Remotion Headless Browser
        run: npx remotion browser ensure

      - name: Render Remotion Composition
        run: |
          mkdir -p out
          npx remotion render "${{ github.event.inputs.composition_id }}" "out/${{ github.event.inputs.output_filename }}" --concurrency=2

      - name: Upload Video Directly to Cloudflare R2
        env:
          AWS_ACCESS_KEY_ID: ${{ secrets.R2_ACCESS_KEY_ID }}
          AWS_SECRET_ACCESS_KEY: ${{ secrets.R2_SECRET_ACCESS_KEY }}
          AWS_DEFAULT_REGION: auto
        run: |
          TIMESTAMP=$(date +"%Y-%m-%d_%H-%M-%S")
          FINAL_NAME="${TIMESTAMP}_${{ github.event.inputs.output_filename }}"
          aws s3 cp "out/${{ github.event.inputs.output_filename }}" "s3://${{ secrets.R2_BUCKET_NAME }}/${FINAL_NAME}" --endpoint-url "${{ secrets.R2_ENDPOINT }}" --region auto
